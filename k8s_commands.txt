to run the project type the below commands:

kubectl apply -f k8s/00-namespace.yaml
kubectl apply -f k8s/01-secrets.yaml
kubectl apply -f k8s/02-rabbitmq.yaml
kubectl apply -f k8s/03-postgres-payment.yaml
kubectl apply -f k8s/04-postgres-notification.yaml
kubectl apply -f k8s/05-payment-config.yaml
kubectl apply -f k8s/06-payment-service.yaml
kubectl apply -f k8s/07-psp-mock-service.yaml
kubectl apply -f k8s/08-notification-service.yaml
kubectl apply -f k8s/09-ingress.yaml



to get pods type:
	kubectl get pods -n rabbitmq-idempotency
	
to get pods details type:
	kubectl describe pods -n rabbitmq-idempotency

to get deployments type:
	kubectl get deployments -n rabbitmq-idempotency
	
to get services type:
	kubectl get services -n rabbitmq-idempotency
	
to check ingress type:
	kubectl get ingress -n rabbitmq-idempotency

to get all components type:
	kubectl get all -n rabbitmq-idempotency
